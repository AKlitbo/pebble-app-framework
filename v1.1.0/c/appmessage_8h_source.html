15// per-channel handler types
16// a store or a face registers only the channels it cares about
17// so the appmessage pipeline stays separate from who reads the data
18// temp is the reading in the user's unit (WEATHER_NO_TEMP when there is none)
19// the pipeline does not know the unit label
20// the face adds "°C" or "°F" from its own setting when it draws
21typedef void (*WeatherHandler)(int temp, const char *condition);
22typedef void (*CoordsHandler)(const char *lat, const char *lon);
23typedef void (*SettingsChangedHandler)(bool time_or_date_changed);
24// extra weather (humidity plus wind plus sunrise and sunset)
25// missing values arrive as -1 or ""
26typedef void (*WeatherExtraHandler)(int humidity, int wind_kmh, const char *wind_dir,
27 const char *sunrise, const char *sunset);
28// daily forecast (uv plus today's high and low plus rain chance)
29// missing values arrive as INT_MIN so a real negative temp is not read as missing
30typedef void (*WeatherForecastHandler)(int uv, int temp_max, int temp_min, int precip_chance);
31// air readings (feels-like plus surface pressure plus dew point)
32// missing values arrive as INT_MIN so a real negative reading is not read as missing
33typedef void (*WeatherAirHandler)(int feels_like, int pressure, int dew_point);
34// the forecast strips (hourly and 7 day) arrive as packed byte arrays
35// the handler owns the wire format so appmessage just hands the raw bytes and their length across
36typedef void (*WeatherForecastStripHandler)(const uint8_t *buf, uint16_t len);
37typedef void (*LocationNameHandler)(const char *name);
38// the watchlist strip arrives as a packed byte array too
39// the handler owns the wire format
40typedef void (*StockStripHandler)(const uint8_t *buf, uint16_t len);
41// the agenda strip arrives as a packed byte array as well
42// the store owns the wire format
43typedef void (*CalendarStripHandler)(const uint8_t *buf, uint16_t len);
44// the custom-colours string can be too big for one persist key
45// so it rides its own handler instead of the settings table
46// inbound the face splits it and stores it
47// outbound the provider rebuilds it into the buffer appmessage hands over
48// so nobody keeps a worst-case buffer alive between sends
49typedef void (*CustomColorsHandler)(const char *combined);
50typedef void (*CustomColorsProvider)(char *out, size_t n);
