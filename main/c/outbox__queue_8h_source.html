83static inline bool outbox_pending(const OutboxQueue *outbox, OutboxKind kind)
84{
85 if (kind == OUTBOX_NONE)
86 {
87 // the empty flight slot and every unused array entry read as OUTBOX_NONE, so without this
88 // an idle outbox would answer that nothing is already pending

193static inline void outbox_hold_failed(OutboxQueue *outbox, OutboxJob job)
194{
195 // a failure reported once the slot is already free comes with no job, and holding it would
196 // queue a send for nothing ahead of the real requests
197 if (job.kind == OUTBOX_NONE || job.retries_left <= 0)
