26// kinds. the weather, stock, calendar, settings, and fresh reply kinds fit in the 6 every face has,
27// with one to spare. phone battery and now playing each add a slot, only on a face that compiles them

103static inline bool outbox_pending(const OutboxQueue *outbox, OutboxKind kind)
104{
105 if (kind == OUTBOX_NONE)
106 {
107 // the empty flight slot and every unused array entry read as OUTBOX_NONE, so without this
108 // an idle outbox would answer that nothing is already pending
109 return false;
110 }
111
112 if (outbox->inflight.kind == kind)
113 {
114 return true;
115 }
116
117 for (int i = 0; i < outbox->queue_len; i++)
118 {
119 if (outbox->queue[i].kind == kind)
120 {
121 return true;
122 }
123 }
124
125 for (int i = 0; i < outbox->failed_len; i++)
126 {
127 if (outbox->failed[i].kind == kind)
128 {
129 return true;
130 }
131 }
132
133 return false;
134}

144static inline bool outbox_push(OutboxQueue *outbox, OutboxKind kind, int retries)
145{
146 if (outbox_pending(outbox, kind) || outbox->queue_len >= OUTBOX_QUEUE_MAX)
147 {
148 return false;

150
151 outbox->queue[outbox->queue_len].kind = kind;

213static inline void outbox_hold_failed(OutboxQueue *outbox, OutboxJob job)
214{
215 // a failure reported once the slot is already free comes with no job, and holding it would
216 // queue a send for nothing ahead of the real requests
217 if (job.kind == OUTBOX_NONE || job.retries_left <= 0)
218 {
219 return;
220 }
221
222 if (outbox->failed_len < OUTBOX_QUEUE_MAX)
223 {
224 outbox->failed[outbox->failed_len].kind = job.kind;
225 outbox->failed[outbox->failed_len].retries_left = job.retries_left - 1;
226 outbox->failed_len++;
227 }
228}

236static inline int outbox_retry_pass(OutboxQueue *outbox)
237{
238 int moved = 0;
239
240 while (outbox->failed_len > 0 && outbox->queue_len < OUTBOX_QUEUE_MAX)
241 {
242 outbox->queue[outbox->queue_len++] = outbox->failed[0];
243 for (int i = 1; i < outbox->failed_len; i++)
244 {
245 outbox->failed[i - 1] = outbox->failed[i];
246 }
247 outbox->failed_len--;
248 moved++;
249 }
250
251 return moved;
252}
