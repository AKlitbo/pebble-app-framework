96static inline bool outbox_pending(const OutboxQueue *outbox, OutboxKind kind)
97{
98 if (kind == OUTBOX_NONE)
99 {
100 // the empty flight slot and every unused array entry read as OUTBOX_NONE, so without this
101 // an idle outbox would answer that nothing is already pending
102 return false;
103 }
104
105 if (outbox->inflight.kind == kind)
106 {
107 return true;
108 }
109
110 for (int i = 0; i < outbox->queue_len; i++)
111 {
112 if (outbox->queue[i].kind == kind)
113 {
114 return true;
115 }
116 }
117
118 for (int i = 0; i < outbox->failed_len; i++)
119 {
120 if (outbox->failed[i].kind == kind)
121 {
122 return true;
123 }
124 }
125
126 return false;
127}
